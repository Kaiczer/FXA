political_actions = {
	ukrainian_unification = {
		icon = generic_speech

		allowed = {
			is_ukrainian_tag = yes
		}

		available = {
			is_ukrainian_tag = yes 
			if = {
				limit = { original_tag = UKR }
				NOT = { country_exists = USR }
				any_state_in = {
					strategic_region = 130
					is_owned_by = UKR
					NOT = { is_owned_by = USR }
				}
			}
			if = {
				limit = { original_tag = USR }
				NOT = { country_exists = UKR }
				any_state_in = {
					strategic_region = 130
					is_owned_by = USR
					NOT = { is_owned_by = UKR }
				}
			}
		}

		ai_will_do = {
			base = 100
			modifier = {
				factor = 100
				is_ukrainian_tag = yes
			}
		}
		cost = 100
		fire_only_once = yes

		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision complete ukrainian_unification"
			if = { # Transfer every Ukrainian state under Allies (Region 130) to ROOT-is_ukrainian_tag
				limit = {
					any_allied_country = {
						any_owned_state = { region = 130 }
					}
				}
				every_country = {
					limit = {
						any_owned_state = {
							region = 130
							NOT = { is_claim_or_core_of_PREV = yes }
						}
					}
					every_owned_state = {
						limit = { 
							region = 130
							NOT = { is_claim_or_core_of_PREV = yes }
							controller = {
								OR = {
									is_ally_with = ROOT
									is_in_faction_with = ROOT
									is_subject_of = ROOT
								}
							} 
						}
						transfer_state_to = ROOT
					}
				}
			}
			if = { # Wipe the USR's existence from the land
				limit = { original_tag = UKR }
				USR = { clear_cores_of_tag_if_possible = yes }
			}
			else_if = { # Same as above but Communism prevails
				limit = { original_tag = USR }
				UKR = { clear_cores_of_tag_if_possible = yes }
				every_state = {
					limit = { 
						region = 130
					}
					remove_claim_and_add_core_of_ROOT = yes
				}
			}
			set_capital = { state = 28 }
			add_political_power = 100
			add_stability = 0.15
			add_war_support = 0.1
		}
	}
}
