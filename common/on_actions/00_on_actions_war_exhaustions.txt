# On actions for War Exhaustion Mechanic
# Implemented by CaptainKaiczer
on_actions = {
	on_monthly = {
		effect = {
			# Checking for war exhaustion growth in every country on monthly basis
			war_exhaustion_monthly_calculation = yes
		}
	}

	#FROM is war target
	on_declare_war = {
		effect = {
			log = "[GetDateText] [Root.GetName] has declared war on [From.GetName]"
			FROM = { 
				set_country_flag = had_war
				# Set up War Exhaustion Mechanic
				if = {
					limit = { 
						NOT = { has_active_mission = war_exhaustion_foreshadow }
					}
					collect_available_data_war_exhaustion = yes
					activate_mission = war_exhaustion_foreshadow
				}
				# else = { war_exhaustion_effects_initiate = yes }
			}
			ROOT = { 
				set_country_flag = had_war
				# Set up War Exhaustion Mechanic
				if = {
					limit = { 
						NOT = { has_active_mission = war_exhaustion_foreshadow }
					}
					collect_available_data_war_exhaustion = yes
					activate_mission = war_exhaustion_foreshadow
				}
				# else = { war_exhaustion_effects_initiate = yes }
			}			
		}
	}

	# Check every fellow that has joined the war of OVERLORD / FACTION_LEADER
	on_war_relation_added = {
		effect = {
			log = "[GetDateText] [Root.GetName] has entered war"
			if = {
				limit = { 
					NOT = { has_active_mission = war_exhaustion_foreshadow }
				}
				# Set up War Exhaustion Mechanic
				collect_available_data_war_exhaustion = yes
				activate_mission = war_exhaustion_foreshadow
			}
			# else = { war_exhaustion_effects_initiate = yes }
		}
	}

	# On Capitulation Immediate / Update Maximum WE Cap
	on_capitulation_immediate = {
		effect = {
			log = "[GetDateText] [Root.GetName] on_capitulation_immediate effect war_exhaustion_set_war_exhaustion_point_cap"
			war_exhaustion_set_war_exhaustion_point_cap = yes
		}
	}

	# On Uncapitulation / Update Maximum WE Cap
	on_uncapitulation = {
		effect = {
			log = "[GetDateText] [Root.GetName] on_uncapitulation effect war_exhaustion_set_war_exhaustion_point_cap"
			war_exhaustion_set_war_exhaustion_point_cap = yes
		}
	}

	# Peace Conference / Update Maximum WE Cap
	on_peaceconference_ended = {
		effect = {
			log = "[GetDateText] [Root.GetName] on_peaceconference_ended effect war_exhaustion_set_war_exhaustion_point_cap"
			war_exhaustion_set_war_exhaustion_point_cap = yes
		}
	}
	
	# On Peace / Update Maximum WE Cap
	on_peace = {
		effect = {
			log = "[GetDateText] [Root.GetName] on_peace effect war_exhaustion_set_war_exhaustion_point_cap"
			war_exhaustion_set_war_exhaustion_point_cap = yes
		}
	}
}
