# scope is the overlord
# requires a temp variable target - the country to be puppeted
puppet_country_without_changing_government = {
	var:target = { get_current_government_type = yes }
	puppet = var:target
	var:target = { restore_previous_government_type = yes }
}

clear_government_flags = {
	clr_country_flag = is_socialist
	clr_country_flag = is_democratic
	clr_country_flag = is_nationalist

	clr_country_flag = is_revolutionary
	clr_country_flag = is_authoritarian
	clr_country_flag = is_reactionary

	clr_country_flag = is_elected
	clr_country_flag = is_dictatorship

	clr_country_flag = is_classic_democratic
	clr_country_flag = is_progressive_democratic
}

set_government_flags = {
	clear_government_flags = yes
	if = {
		limit = { has_government = authoritarian_socialism }
		set_country_flag = is_socialist
		set_country_flag = is_authoritarian
		set_country_flag = is_dictatorship
	}
	else_if = {
		limit = {
			OR = {
				has_government = vanguard_socialism
				has_government = revolutionary_socialism
			}
		}
		set_country_flag = is_socialist
		set_country_flag = is_revolutionary
	}
	else_if = {
		limit = {
			OR = {
				has_government = paternalism
				has_government = revolutionary_nationalism
				has_government = volkism
				has_government = despotism
				has_government = ultrareactionism
			}
		}
		set_country_flag = is_nationalist
	}
	else_if = {
		limit = {
			OR = {
				has_government = volkism
				has_government = despotism
				has_government = ultrareactionism
			}
		}
		set_country_flag = is_dictatorship
	}
	else_if = {
		limit = {
			OR = {
				has_government = paternalism
				has_government = volkism
			}
		}
		set_country_flag = is_authoritarian
	}
	else_if = {
		limit = {
			has_government = revolutionary_nationalism
		}
		set_country_flag = is_revolutionary
	}
	else_if = {
		limit = {
			OR = {
				has_government = despotism
				has_government = ultrareactionism
			}
		}
		set_country_flag = is_reactionary
	}
	else = {
		set_country_flag = is_elected
		set_country_flag = is_democratic
		if = {
			limit = {
				OR = {
					has_government = social_conservatism
					has_government = classical_liberalism
				}
			}
			set_country_flag = is_classic_democratic
		}
		else = {
			set_country_flag = is_progressive_democratic
		}
	}
}

set_default_government = {
	if = {
		limit = {
			has_elections = no
		}
		add_ideas = {
			unitary_state
			paternal_dictatorship
			executive_hos
			rule_by_decree
			manipulated_judiciary
			procrastinated_bureaucracy
			all_bow_to_leader
			cultural_hegemony
		}
	}
	if = {
		limit = {
			is_monarchy = yes
		}
		add_ideas = {
			unitary_state
			semi_constitutional_monarchy
			joint_executive
			well_functioned_parliament
			well_functioned_courts
			centralized_bureaucracy
			legality_equilibrium
			homogeneous_culture
		}
	}
	if = {
		limit = {
			has_socialist_government = yes
		}
		add_ideas = {
			union_of_soviets
			soviet_republic
			partylaw
			peoples_legislature
			well_functioned_courts
			parallel_bureaucracy
			all_bow_to_law
			cultural_autonomy
		}
	}
	else = {
		add_ideas = {
			unitary_state
			presidential_republic
			executive_hos
			well_functioned_parliament
			well_functioned_courts
			centralized_bureaucracy
			legality_equilibrium
			cultural_conservatism
		}
	}
}
