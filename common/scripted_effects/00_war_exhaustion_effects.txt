### Initialization / Update ###
collect_available_data_war_exhaustion = {
	war_exhaustion_collect_manpower_data = yes
	war_exhaustion_localisation_variable = yes
}

war_exhaustion_effects_initiate = {
	war_exhaustion_set_war_exhaustion_point_cap = yes
	set_up_war_exhaustion_effects_start = yes
}

war_exhaustion_effects_update = {
	war_exhaustion_level_update = yes
	war_exhaustion_penalty_effects_update = yes
	if = {
		limit = { 
			has_war = no
			check_variable = { war_exhaustion_modifier_level = 1 }
		}
		reset_war_exhaustion_effects = yes
	}
}

### Actual Calculation / Configuration ###
set_up_war_exhaustion_effects_start = {
	# All decimal variables
	set_variable = { war_exhaustion_pp_factor_var = -0.002 }
	set_variable = { war_exhaustion_stability_factor_var = -0.025 }
	set_variable = { war_exhaustion_war_support_factor_var = -0.04 }
	set_variable = { war_exhaustion_consumer_goods_factor_var = 0.015 }
	set_variable = { war_exhaustion_industrial_capacity_factor_var = -0.02 }
	set_variable = { war_exhaustion_local_resources_factor_var = -0.015 }
	set_variable = { war_exhaustion_production_speed_factor_var = -0.025 }
	set_variable = { war_exhaustion_industrial_repair_factor_var = -0.02 }
	set_variable = { war_exhaustion_army_org_factor_var = -0.015 }
	set_variable = { war_exhaustion_land_reinforce_rate_var = -0.02 }
	#Only Non-decimal variable (max: 800)
	if = {
		limit = {
			check_variable = { war_exhaustion_modifier_level > 3 }
		}
		set_variable = {  war_exhaustion_weekly_manpower_var = -200 }
	}
}

reset_war_exhaustion_effects = {
	# Reset all modifiers to 0 when at peace and LV.1
	set_variable = { war_exhaustion_pp_factor_var = 0 }
	set_variable = { war_exhaustion_stability_factor_var = 0 }
	set_variable = { war_exhaustion_war_support_factor_var = 0 }
	set_variable = { war_exhaustion_consumer_goods_factor_var = 0 }
	set_variable = { war_exhaustion_industrial_capacity_factor_var = 0 }
	set_variable = { war_exhaustion_local_resources_factor_var = 0 }
	set_variable = { war_exhaustion_production_speed_factor_var = 0 }
	set_variable = { war_exhaustion_industrial_repair_factor_var = 0 }
	set_variable = { war_exhaustion_army_org_factor_var = 0 }
	set_variable = { war_exhaustion_land_reinforce_rate_var = 0 }
	set_variable = { war_exhaustion_weekly_manpower_var = 0 }
}

war_exhaustion_level_update = {
	if = {
		limit = { 
			check_variable = { 
				var = war_exhaustion_point_actual
				value = 175
				compare = greater_than_or_equals
			}
		}
		set_variable = { war_exhaustion_modifier_level = 8 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual > 149 }
			check_variable = { war_exhaustion_point_actual < 175 }
		}
		set_variable = { war_exhaustion_modifier_level = 7 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual > 124 }
			check_variable = { war_exhaustion_point_actual < 150 }
		}
		set_variable = { war_exhaustion_modifier_level = 6 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual > 101 }
			check_variable = { war_exhaustion_point_actual < 124 }
		}
		set_variable = { war_exhaustion_modifier_level = 5 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual > 74 }
			check_variable = { war_exhaustion_point_actual < 101 }
		}
		set_variable = { war_exhaustion_modifier_level = 4 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual > 49 }
			check_variable = { war_exhaustion_point_actual < 75 }
		}
		set_variable = { war_exhaustion_modifier_level = 3 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual > 24 }
			check_variable = { war_exhaustion_point_actual < 50 }
		}
		set_variable = { war_exhaustion_modifier_level = 2 }
	}
	else = {
		set_variable = { war_exhaustion_modifier_level = 1 }
	}
}

war_exhaustion_penalty_effects_update = {
	# Reset to Base Value First
	set_up_war_exhaustion_effects_start = yes 
	
	# Apply the modifier level to the base factors to get the actual penalties
	set_temp_variable = { war_exhaustion_multiplier = war_exhaustion_modifier_level }
	multiply_temp_variable = { war_exhaustion_multiplier = 2.5 }
	
	# Apply the multiplier to all penalty variables
	multiply_variable = { war_exhaustion_pp_factor_var = war_exhaustion_multiplier }
	multiply_variable = { war_exhaustion_stability_factor_var = war_exhaustion_multiplier }
	multiply_variable = { war_exhaustion_war_support_factor_var = war_exhaustion_multiplier }
	multiply_variable = { war_exhaustion_consumer_goods_factor_var = war_exhaustion_multiplier }
	multiply_variable = { war_exhaustion_industrial_capacity_factor_var = war_exhaustion_multiplier }
	multiply_variable = { war_exhaustion_local_resources_factor_var = war_exhaustion_multiplier }
	multiply_variable = { war_exhaustion_production_speed_factor_var = war_exhaustion_multiplier }
	multiply_variable = { war_exhaustion_industrial_repair_factor_var = war_exhaustion_multiplier }
	multiply_variable = { war_exhaustion_army_org_factor_var = war_exhaustion_multiplier }
	multiply_variable = { war_exhaustion_land_reinforce_rate_var = war_exhaustion_multiplier }
	# Manpower penalty is calculated differently, as it's a flat value rather than a percentage. It starts at -200 and goes up to -800 at max exhaustion.
	multiply_variable = { war_exhaustion_weekly_manpower_var = war_exhaustion_modifier_level }
}

war_exhaustion_collect_manpower_data = {
	# Find 30% of Available Manpower 
	clear_variable = casualties_threshold # For Calculation Only
	set_temp_variable = { threshold_temp = 0 }
	if = {
		limit = { has_manpower > 9999 }
		add_to_temp_variable = { threshold_temp = max_available_manpower_k }
	}
	else = { 
		add_to_temp_variable = { threshold_temp = max_available_manpower }
	}
	
	divide_temp_variable = { threshold_temp = 10 }
	multiply_temp_variable = { threshold_temp = 3 }

	set_variable = { casualties_threshold = threshold_temp }
}

war_exhaustion_collect_factory_data = {
	# Find CIV Factories in Every Owned States
	clear_variable = total_civilian_factories # For Calculation Only
	set_variable = { total_civilian_factories = num_of_civilian_factories@THIS.var:owned_states }

	# Find MIL Factories in Every Owned States
	clear_variable = total_military_factories # For Calculation Only
	set_variable = { total_military_factories = num_of_military_factories@THIS.var:owned_states }
}

# Core Occupier > Nearest > Others
war_exhaustion_set_war_exhaustion_point_cap = {
	if = {
		limit = { has_war = yes }
		if = {
			limit = {
				OR = {
					any_core_state = {
						is_controlled_by_enemies = yes 
						is_in_home_area = yes
					}
					surrender_progress > 0.15
					any_enemy_country = { 
						OR = {
							is_actual_major = yes
							is_in_faction_with_major = yes
							AND = {
								is_faction_leader = yes
								num_faction_members > 1
							}
						}
					}
				}
			}
			set_variable = { maximum_war_exhaustion_cap = 200 }
		}
		else_if = {
			limit = {
				all_enemy_country = { #Every general enemy
					OR = {
						is_faction_leader = no
						AND = {
							is_faction_leader = yes
							num_faction_members < 2
						}
					}
					is_neighbor_of = ROOT
					is_actual_major = no
				}
				enemies_strength_ratio > 1.3
			}
			set_variable = { maximum_war_exhaustion_cap = 150 }
		}
		else = {
			set_variable = { maximum_war_exhaustion_cap = 100 }
		}
	}
	else = { set_variable = { maximum_war_exhaustion_cap = 0 } }
}

war_exhaustion_localisation_variable = {
	clear_variable = casualties_threshold_loc # For Localisation Only

	set_variable = { casualties_threshold_loc = casualties_threshold } # Set Localisation
	
	# IF casualties_threshold reached 1000K (1 Million)
	if = {
		limit = {
			check_variable = { 
				var = casualties_threshold	
				value = 1000
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { temp = casualties_threshold }
		divide_temp_variable = { temp = 1000 }
		set_variable = { casualties_threshold_loc = temp }
	}
}

war_exhaustion_monthly_calculation = {
	set_variable = { exhaustion_growth = 0 }
	# monthly_growth 
	add_to_variable = { exhaustion_growth = 1.005 }
	# casualties 
	if = {
		limit = {
			check_variable = { 
				var = casualties_k
				value = 100
				compare = less_than_or_equals
			}
		}
		add_to_variable = { exhaustion_growth = 0.045 }
	}
	else_if = {
		limit = {
			check_variable = { casualties_k > 100 }
			check_variable = { casualties_k < 2500 }
		}
		add_to_variable = { exhaustion_growth = 0.075 }
	}
	else_if = {
		limit = {
			check_variable = { 
				var = casualties_k
				value = 2500
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { exhaustion_growth = 0.1 }
	}
	# damaged factories
	war_exhaustion_collect_factory_data = yes
	if = {
		limit = {
			any_owned_state = {
				check_variable = { damaged_building_level@industrial_complex > 0 }
			}
		}
		add_to_variable = { total_damaged_civilian_factories = damaged_building_level@industrial_complex@THIS.var:owned_states }
		divide_variable = { total_damaged_civilian_factories = total_civilian_factories }
		multiply_variable = { total_damaged_civilian_factories = 100 }
		if = {
			limit = { check_variable = { total_damaged_civilian_factories > 9 } }
			add_to_variable = { exhaustion_growth = 0.1 }
		}
		else_if = {
			limit = { check_variable = { total_damaged_civilian_factories > 19 } }
			add_to_variable = { exhaustion_growth = 0.2 }
		}
		else_if = {
			limit = { check_variable = { total_damaged_civilian_factories > 29 } }
			add_to_variable = { exhaustion_growth = 0.3 }
		}
		else_if = {
			limit = { check_variable = { total_damaged_civilian_factories > 39 } }
			add_to_variable = { exhaustion_growth = 0.4 }
		}
	}
	if = {
		limit = {
			any_owned_state = {
				check_variable = { damaged_building_level@arms_factory > 0 }
			}
		}
		add_to_variable = { total_damaged_military_factories = damaged_building_level@arms_factory@THIS.var:owned_states }
		divide_variable = { total_damaged_military_factories = total_military_factories }
		multiply_variable = { total_damaged_military_factories = 100 }
		if = {
			limit = { check_variable = { total_damaged_military_factories > 9 } }
			add_to_variable = { exhaustion_growth = 0.1 }
		}
		else_if = {
			limit = { check_variable = { total_damaged_military_factories > 19 } }
			add_to_variable = { exhaustion_growth = 0.2 }
		}
		else_if = {
			limit = { check_variable = { total_damaged_military_factories > 29 } }
			add_to_variable = { exhaustion_growth = 0.3 }
		}
		else_if = {
			limit = { check_variable = { total_damaged_military_factories > 39 } }
			add_to_variable = { exhaustion_growth = 0.4 }
		}
	}
	# conscription
	if = {
		limit = { has_idea = low_economic_mobilisation }
		add_to_variable = { exhaustion_growth = 0.2 }
	}
	else_if = {
		limit = { has_idea = partial_economic_mobilisation }
		add_to_variable = { exhaustion_growth = 0.4 }
	}
	else_if = {
		limit = { has_idea = war_economy }
		add_to_variable = { exhaustion_growth = 0.6 }
	}
	else_if = {
		limit = { has_idea = total_economic_mobilisation }
		add_to_variable = { exhaustion_growth = 0.8 }
	}
	else_if = {
		limit = {
			OR = {
				has_idea = undisturbed_isolation
				has_idea = isolation
				has_idea = civilian_economy
			}
			has_war = no
		}
		add_to_variable = { exhaustion_growth = -0.3 }
	}
	else = { add_to_variable = { exhaustion_growth = 0.2 } }
	# surrender_progress
	if = {
		limit = { surrender_progress > 0 }
		set_temp_variable = { surrender_point_temp = 5 }
		multiply_temp_variable = { surrender_point_temp = surrender_progress }
		add_to_variable = { exhaustion_growth = surrender_point_temp }
	}
	# peace_status
	if = {
		limit = { 
			has_war = no
			check_variable = { war_exhaustion_point_actual > 0 }
		}
		if = {
			limit = {
				check_variable = { maximum_war_exhaustion_cap = 100 }
			}
			add_to_variable = { exhaustion_growth = -5 }
		}
		else_if = {
			limit = {
				check_variable = { maximum_war_exhaustion_cap = 150 }
			}
			add_to_variable = { exhaustion_growth = -7 }
		}
		else_if = {
			limit = {
				check_variable = { maximum_war_exhaustion_cap = 200 }
			}
			add_to_variable = { exhaustion_growth = -10 }
		}
	}
	set_variable = { war_exhaustion_point_growth = exhaustion_growth }
	add_to_variable = { war_exhaustion_point_actual = exhaustion_growth }
	clamp_variable = {
		var = war_exhaustion_point_actual
		min = 0
		max = maximum_war_exhaustion_cap
	}
}
