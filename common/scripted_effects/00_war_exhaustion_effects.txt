war_exhaustion_level_update = {
	if = {
		limit = { 
			check_variable = { 
				var = war_exhaustion_point_actual
				value = 175
				compare = greater_than_or_equals
			}
		}
		set_variable = { war_exhaustion_modifier_level = 8 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual = 150 }
			check_variable = { war_exhaustion_point_actual < 175 }
		}
		set_variable = { war_exhaustion_modifier_level = 7 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual = 125 }
			check_variable = { war_exhaustion_point_actual < 150 }
		}
		set_variable = { war_exhaustion_modifier_level = 6 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual = 101 }
			check_variable = { war_exhaustion_point_actual < 124 }
		}
		set_variable = { war_exhaustion_modifier_level = 5 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual = 75 }
			check_variable = { war_exhaustion_point_actual < 101 }
		}
		set_variable = { war_exhaustion_modifier_level = 4 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual = 50 }
			check_variable = { war_exhaustion_point_actual < 75 }
		}
		set_variable = { war_exhaustion_modifier_level = 3 }
	}
	else_if = {
		limit = { 
			check_variable = { war_exhaustion_point_actual = 25 }
			check_variable = { war_exhaustion_point_actual < 50 }
		}
		set_variable = { war_exhaustion_modifier_level = 2 }
	}
	else = {
		set_variable = { war_exhaustion_modifier_level = 1 }
	}
}

war_exhaustion_collect_manpower_data = {
	# Find 30% of Available Manpower 
	clear_variable = casualties_threshold # For Calculation Only
	set_temp_variable = { threshold_temp = 0 }
	if = {
		limit = { has_manpower > 9999 }
		add_to_temp_variable = { threshold_temp = max_available_manpower_k }
	}
	else = { 
		add_to_temp_variable = { threshold_temp = max_available_manpower }
	}
	
	divide_temp_variable = { threshold_temp = 10 }
	multiply_temp_variable = { threshold_temp = 3 }

	set_variable = { casualties_threshold = threshold_temp }
}

war_exhaustion_set_war_exhaustion_point_cap = {
	if = {
		limit = { has_war = yes }
		if = {
			limit = {
				OR = {
					any_core_state = {
						is_controlled_by_enemies = yes 
						is_in_home_area = yes
					}
					surrender_progress > 0.15
					any_enemy_country = { 
						OR = {
							is_actual_major = yes
							is_in_faction_with_major = yes
							AND = {
								is_faction_leader = yes
								num_faction_members > 1
							}
						}
					}
				}
			}
			set_variable = { maximum_war_exhaustion_cap = 200 }
		}
		else_if = {
			limit = {
				all_enemy_country = { #Every general enemy
					OR = {
						is_faction_leader = no
						AND = {
							is_faction_leader = yes
							num_faction_members < 2
						}
					}
					is_neighbor_of = ROOT
					is_in_faction_with_major = no
					is_actual_major = no
				}
				enemies_strength_ratio > 1.3
			}
			set_variable = { maximum_war_exhaustion_cap = 150 }
		}
		else = {
			set_variable = { maximum_war_exhaustion_cap = 100 }
		}
	}
	else = { set_variable = { maximum_war_exhaustion_cap = 0 } }
}

war_exhaustion_localisation_variable = {
	clear_variable = casualties_threshold_loc # For Localisation Only

	set_variable = { casualties_threshold_loc = casualties_threshold } # Set Localisation
	
	# IF casualties_threshold reached 1000K (1 Million)
	if = {
		limit = {
			check_variable = { 
				var = casualties_threshold	
				value = 1000
				compare = greater_than_or_equals
			}
		}
		set_temp_variable = { temp = casualties_threshold }
		divide_temp_variable = { temp = 1000 }
		set_variable = { casualties_threshold_loc = temp }
	}
}
