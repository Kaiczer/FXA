# checks for country in scope of three socialist-based ideologies
has_socialist_government = {
	custom_override_tooltip = {
		tooltip = {
			localization_key = key_loc_has_government
			IDEOLOGY = {
				localization_key = key_loc_punc_comma
				ITEM = authoritarian_socialism
				NEXT_ITEM = {
					localization_key = key_loc_grammar_or
					ITEM = vanguard_socialism
					NEXT_ITEM = revolutionary_socialism
				}
			}
		}
		has_country_flag = is_socialist
	}
}

# checks whether the country in scope has the same government type as ROOT (the country running the event/focus)
# note that it's not the exact same government - just the general type, as above
same_government_type_as_ROOT = {
	OR = {
		AND = {
			has_socialist_government = yes
			ROOT = { has_socialist_government = yes }
		}
		AND = {
			has_democratic_government = yes
			ROOT = { has_democratic_government = yes }
		}
		AND = {
			has_authoritarian_government = yes
			ROOT = { has_authoritarian_government = yes }
		}
	}
}

same_government_type_as_ROOT_paternalism = {
	OR = {
		AND = {
			has_socialist_government = yes
			ROOT = { has_socialist_government = yes }
		}
		AND = {
			has_elected_government = yes
			ROOT = { has_elected_government = yes }
		}
		AND = {
			has_any_authoritarian_government = yes
			ROOT = { has_any_authoritarian_government = yes }
		}
	}
}

both_socialist_or_both_not_socialist = {
	OR = {
		AND = {
			has_socialist_government = yes
			PREV = { has_socialist_government = yes }
		}
		AND = {
			has_socialist_government = no
			PREV = { has_socialist_government = no }
		}
	}
}

is_monarchy = {
	custom_trigger_tooltip = {
		tooltip = is_monarchy
		has_socialist_government = no 
		any_character = {
			is_country_leader = yes
			OR = {
				has_character_flag = is_monarch
				has_character_flag = is_regent
			}
		}
	}
}

is_warlord_state = {
	custom_trigger_tooltip = {
		tooltip = is_warlord_state
		any_character = {
			is_country_leader = yes
			has_character_flag = is_warlord
		}
	}
}
